cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES x86_64)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
project(mkw_macos_probe LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin" OR NOT CMAKE_HOST_SYSTEM_PROCESSOR STREQUAL "x86_64")
  message(FATAL_ERROR "G1 requires a native x86_64 Mac")
endif()

set(SDL3_FRAMEWORK_DIR "" CACHE PATH "Directory containing the macOS SDL3.framework")
set(VULKAN_PREFIX "/usr/local" CACHE PATH "LunarG SDK prefix containing include/ and lib/")
find_library(SDL3_FRAMEWORK SDL3 PATHS "${SDL3_FRAMEWORK_DIR}" NO_DEFAULT_PATH REQUIRED)
find_path(VULKAN_INCLUDE vulkan/vulkan.h PATHS "${VULKAN_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_file(VULKAN_LOADER libvulkan.1.dylib PATHS "${VULKAN_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)

add_executable(mkw-macos-probe main.c device.c frames.c)
target_compile_features(mkw-macos-probe PRIVATE c_std_11)
target_compile_options(mkw-macos-probe PRIVATE -Wall -Wextra -Werror)
target_include_directories(mkw-macos-probe PRIVATE "${VULKAN_INCLUDE}")
target_link_libraries(mkw-macos-probe PRIVATE "${SDL3_FRAMEWORK}" "${VULKAN_LOADER}")
set_target_properties(mkw-macos-probe PROPERTIES BUILD_RPATH "${SDL3_FRAMEWORK_DIR};${VULKAN_PREFIX}/lib")

enable_testing()
add_test(NAME macos-nvk-wsi COMMAND mkw-macos-probe)
set_tests_properties(macos-nvk-wsi PROPERTIES TIMEOUT 60)
