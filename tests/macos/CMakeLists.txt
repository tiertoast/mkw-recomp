cmake_minimum_required(VERSION 3.24)
project(mkw_darwin_foundations LANGUAGES C CXX ASM)
if(NOT APPLE OR NOT CMAKE_C_COMPILER_ID STREQUAL "AppleClang" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64"
   OR NOT MKW_GAME_ID OR NOT MKW_HOST_EXPORTS)
  message(FATAL_ERROR "Configure via tools/build/build.py --target macos-x86_64")
endif()

set(rt "${CMAKE_CURRENT_LIST_DIR}/../../runtime")
if(MKW_FRAME_POINTERS)
  add_compile_options(-fno-omit-frame-pointer)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "@loader_path")
add_compile_options(-Wall -Wextra -Werror -march=x86-64-v3)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-stdlib=libc++>")
add_link_options(-Wl,-dead_strip "$<$<LINK_LANGUAGE:CXX>:-stdlib=libc++>")
add_compile_definitions(RT_GAME_ID="${MKW_GAME_ID}")
include_directories("${rt}" "${rt}/core/base" "${rt}/core/boot" "${rt}/core/diag" "${rt}/cpu")

include("${CMAKE_CURRENT_LIST_DIR}/../../tools/build/darwin.cmake")
add_library(mkwgame SHARED module.c)
set_target_properties(mkwgame PROPERTIES PREFIX "" INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON C_VISIBILITY_PRESET hidden)
target_link_options(mkwgame PRIVATE "LINKER:-exported_symbol,_mkwgame_tables")
target_link_libraries(mkwgame PRIVATE mkw-host)

add_library(registrations STATIC declarations.cpp)
add_executable(foundations fixture.c support.c "${rt}/cpu/hostfn.c" "${rt}/cpu/dispatch.c" "${rt}/cpu/sjlj.c"
               "${rt}/core/boot/gamelib.c")
set_target_properties(foundations PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
target_link_libraries(foundations PRIVATE darwin-platform mkw-host "$<LINK_LIBRARY:WHOLE_ARCHIVE,registrations>")
add_dependencies(foundations mkwgame)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/mkwgame.id" CONTENT "${MKW_GAME_ID}")

enable_testing()
add_executable(thread-test threads.c)
target_link_libraries(thread-test PRIVATE darwin-platform)
add_test(NAME darwin-threads COMMAND thread-test)
add_test(NAME macho-roundtrip COMMAND foundations "$<TARGET_FILE_DIR:mkwgame>")
set_tests_properties(darwin-threads macho-roundtrip PROPERTIES TIMEOUT 20)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME macho-contract COMMAND "${Python3_EXECUTABLE}" -I "${CMAKE_CURRENT_SOURCE_DIR}/check.py"
         "$<TARGET_FILE:foundations>" "${MKW_GAME_ID}" "${MKW_HOST_EXPORTS}")
set_tests_properties(macho-contract PROPERTIES TIMEOUT 30)

# No entries of any kind. Exercise ld's synthesized section bounds with the real lookup code.
add_executable(empty-registry empty.c "${rt}/cpu/hostfn.c")
add_test(NAME macho-empty-registry COMMAND empty-registry)
set_tests_properties(macho-empty-registry PROPERTIES TIMEOUT 10)

add_library(vm-fixture STATIC vm_layout.c "${rt}/core/memory/memory.c")
target_link_libraries(vm-fixture PUBLIC darwin-platform)
target_include_directories(vm-fixture PUBLIC "${rt}/core/memory")
add_executable(vm-test vm.c)
target_link_libraries(vm-test PRIVATE vm-fixture)
add_test(NAME darwin-vm COMMAND vm-test)
add_test(NAME darwin-vm-reexec COMMAND vm-test retry)
add_test(NAME darwin-vm-faults COMMAND "${Python3_EXECUTABLE}" -I "${CMAKE_CURRENT_SOURCE_DIR}/faults.py" "$<TARGET_FILE:vm-test>")
set_tests_properties(darwin-vm darwin-vm-reexec darwin-vm-faults PROPERTIES TIMEOUT 30)

option(MKW_TEST_NVK "Build the guest VM upload/readback test for the NullMoth NVK ICD" OFF)
if(MKW_TEST_NVK)
  set(VULKAN_PREFIX "/usr/local" CACHE PATH "LunarG SDK prefix containing include/ and lib/")
  find_path(VULKAN_INCLUDE vulkan/vulkan.h PATHS "${VULKAN_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_file(VULKAN_LOADER libvulkan.1.dylib PATHS "${VULKAN_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  add_executable(vm-vulkan vm_vulkan.c)
  target_include_directories(vm-vulkan PRIVATE "${VULKAN_INCLUDE}")
  target_link_libraries(vm-vulkan PRIVATE vm-fixture "${VULKAN_LOADER}")
  set_target_properties(vm-vulkan PROPERTIES INSTALL_RPATH "${VULKAN_PREFIX}/lib")
  add_test(NAME darwin-vm-nvk COMMAND vm-vulkan)
  set_tests_properties(darwin-vm-nvk PROPERTIES TIMEOUT 60)
endif()

add_executable(scheduler-test scheduler.c support.c "${rt}/core/memory/memory.c" "${rt}/core/base/cfg.c"
               "${rt}/core/diag/perf.c" "${rt}/core/boot/gamelib.c" "${rt}/cpu/dispatch.c" "${rt}/cpu/hostfn.c"
               "${rt}/cpu/sjlj.c")
target_include_directories(scheduler-test PRIVATE "${rt}/core/kernel" "${rt}/gfx/nvn/host" "${rt}/gfx/nvn/api" "${rt}/gfx/nvn/procs")
target_link_libraries(scheduler-test PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,darwin-kernel>" mkw-host)
set_target_properties(scheduler-test PROPERTIES C_VISIBILITY_PRESET hidden)
add_test(NAME darwin-scheduler COMMAND scheduler-test scheduler)
add_test(NAME darwin-guest-fiber COMMAND scheduler-test fiber)
set_tests_properties(darwin-scheduler darwin-guest-fiber PROPERTIES TIMEOUT 12
                     ENVIRONMENT "MKW_ASYNC_PREEMPT=2;MKW_FREE_CORES=0;MKW_CORE_NUM_GATE=0;MKW_SPIN_HANDOFF_US=0")
set_property(TEST darwin-guest-fiber APPEND PROPERTY ENVIRONMENT "MKW_CORE_DIAG=1")

add_executable(fiber-test fibers.c fiber_regs.S)
target_link_libraries(fiber-test PRIVATE darwin-platform)
add_test(NAME darwin-fiber-context COMMAND fiber-test)
add_test(NAME darwin-fiber-guard COMMAND fiber-test guard)
set_tests_properties(darwin-fiber-context darwin-fiber-guard PROPERTIES TIMEOUT 8)
set_tests_properties(darwin-fiber-context darwin-fiber-guard darwin-guest-fiber PROPERTIES SKIP_RETURN_CODE 77)

add_executable(darwin-file-test files.c)
target_link_libraries(darwin-file-test PRIVATE darwin-platform)
add_test(NAME darwin-files-and-log COMMAND darwin-file-test files)
add_test(NAME darwin-atomic-saves COMMAND darwin-file-test saves)
set_tests_properties(darwin-files-and-log darwin-atomic-saves PROPERTIES TIMEOUT 15)

add_executable(darwin-services-test services.c)
target_link_libraries(darwin-services-test PRIVATE darwin-platform)
add_dependencies(darwin-services-test mkw-host)
add_test(NAME darwin-time-locale COMMAND darwin-services-test time -AppleLanguages "(fr-CA)"
         -AppleLocale "en_IT@calendar=gregorian")
add_test(NAME darwin-process-loading COMMAND darwin-services-test process "$<TARGET_FILE:mkw-host>" "argument with spaces")
set_tests_properties(darwin-time-locale darwin-process-loading PROPERTIES TIMEOUT 12)
